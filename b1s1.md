### Scala (scala)

val nums = List(1, 2, 3, 4, 5)

nums.map(n => (n, n * n))

### Python (ipython)

```python
lst = [1, 2, 3, 4, 5]
list(map(lambda n: n * 10, lst))
list(map(lambda n: (n, n * n), lst))
list(filter(lambda n : n % 2 == 0, lst))
from functools import reduce
reduce(lambda x, y : x * y, lst, 1)
```
